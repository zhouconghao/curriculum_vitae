% Venue text: "Phys. Rev. D 110, 103508 (2024)", "arXiv:2606.21141 (2026)" or "LSST Technote SITCOMTN-165 (2025)"
\newbibmacro*{cv:venue:text}{%
  \ifboolexpr{ test {\iffieldundef{journaltitle}} or test {\iffieldequalstr{journaltitle}{arXiv e-prints}} }
    {\iffieldundef{eprint}
       {\thefield{type}\iffieldundef{number}{}{~\thefield{number}}}
       {arXiv:\thefield{eprint}}}
    {{\frenchspacing % no sentence-end spacing after the dots in "Mon. Not. R. Astron. Soc."
      \iffieldundef{shortjournal}{\thefield{journaltitle}}{\thefield{shortjournal}}}%
     \iffieldundef{volume}{}{~\iffieldsequal{volume}{year}{\thefield{number}}{\thefield{volume}}}%
     \iffieldundef{pages}{}{, \usefield{\mkfirstpage}{pages}}}%
  \iffieldundef{year}{}{~(\thefield{year})}}

% Link it to the DOI, else arXiv, else the entry's url
\newbibmacro*{cv:link}[1]{%
  \iffieldundef{doi}
    {\iffieldundef{eprint}
       {\iffieldundef{url}{#1}{\href{\thefield{url}}{#1}}}
       {\href{https://arxiv.org/abs/\thefield{eprint}}{#1}}}
    {\href{https://doi.org/\thefield{doi}}{#1}}}

% Journal papers also get their arXiv number: "Phys. Rev. D 110, 103508 (2024) [arXiv:2312.11789]"
\newbibmacro*{cv:arxiv}{%
  \ifboolexpr{ test {\iffieldundef{eprint}} or test {\iffieldundef{journaltitle}} or
               test {\iffieldequalstr{journaltitle}{arXiv e-prints}} }
    {}
    {\setunit{\addspace}%
     \printtext[brackets]{\href{https://arxiv.org/abs/\thefield{eprint}}{arXiv:\thefield{eprint}}}}}

\newbibmacro*{cv:venue}{%
  \ifboolexpr{ test {\iffieldundef{journaltitle}} and test {\iffieldundef{eprint}} and test {\iffieldundef{type}} }
    {}
    {\printtext{\usebibmacro{cv:link}{\usebibmacro{cv:venue:text}}}}%
  \usebibmacro{cv:arxiv}%
  \newunit
  \printfield{pubstate}}

\newbibmacro*{cv:entry}{%
  \printfield{title}%
  \setunit{\newline}% journal and arXiv on their own line, so the reference never splits
  \usebibmacro{cv:venue}%
  \newunit
  \printfield{note}%
  \setunit{\newline}%
  \printnames{author}%
  \finentry}
